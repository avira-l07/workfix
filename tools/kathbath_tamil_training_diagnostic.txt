def command(args, name):
    token = os.environ.get('HF_TOKEN') or '<no-token>'
    path = logs / (name + '.log')
    with path.open('w', encoding='utf-8') as log:
        process = subprocess.Popen(list(map(str, args)), stdout=subprocess.PIPE,
                                   stderr=subprocess.STDOUT, text=True, bufsize=1,
                                   env=dict(os.environ, PYTHONUNBUFFERED='1', PYTHONFAULTHANDLER='1'))
        for line in process.stdout:
            clean = line.replace(token, '[REDACTED]')
            log.write(clean)
            log.flush()
            print(clean, end='', flush=True)
        returncode = process.wait()
        summary.setdefault('process_exit_codes', {})[name] = returncode
        footer = f'\n[process_exit] {name}: returncode={returncode}\n'
        log.write(footer)
        print(footer, end='', flush=True)
        if returncode:
            raise RuntimeError(f'{name} failed (returncode={returncode}); see {path.name} and the report ZIP')

def retry_first_pilot():
    command(['nvidia-smi', '--query-gpu=name,memory.used,memory.total', '--format=csv'], 'training-diagnostic-gpu')
    memory = Path('/proc/meminfo')
    if memory.is_file():
        summary['training_host_memory_before_retry'] = [line for line in memory.read_text().splitlines()
                        if line.startswith(('MemTotal:', 'MemAvailable:', 'SwapTotal:', 'SwapFree:'))]
    def cgroup_memory():
        return {name: (Path('/sys/fs/cgroup') / name).read_text().strip()
                for name in ('memory.max', 'memory.current', 'memory.events')
                if (Path('/sys/fs/cgroup') / name).is_file()}
    summary['training_cgroup_memory_before_retry'] = cgroup_memory()
    if shutil.disk_usage(work).free < 3 * 1024**3:
        raise RuntimeError('Too little free disk for the pilot checkpoint; preserve reports')
    if sum(p.stat().st_size for p in work.rglob('*') if p.is_file()) > 16 * 1024**3:
        raise RuntimeError('Pilot workspace exceeds the reviewed 16 GiB output bound')
    try:
        command([train_python, work / 'kathbath_tamil_pilot.py', 'pilot', '--repo', work,
                 '--mode', 'freeze-lower', '--seed', '17'], 'pilot-freeze-lower-17-diagnostic')
    finally:
        summary['training_cgroup_memory_after_retry'] = cgroup_memory()
    summary['status'] = 'First pilot retry completed; other seeds/modes and full trial not completed'

stage('first-pilot-diagnostic', retry_first_pilot)
